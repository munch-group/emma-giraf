# Opdater sfs_to_tmrca med n_sub parameter
def allele_counts_to_sfs_obs(df, n_sub=8, n_obs=5000, seed=42):
    rng = np.random.default_rng(seed)
    der = df['derived_count'].values.astype(int)
    tot = df['nr_samples'].values.astype(int)
    obs = []
    for idx in rng.integers(0, len(df), n_obs * 20):
        if len(obs) >= n_obs:
            break
        n, ac = tot[idx], der[idx]
        if n < n_sub:
            continue
        k = rng.hypergeometric(ac, n - ac, n_sub)
        if 0 < k < n_sub:
            obs.append(float(k))
    return np.array(obs[:n_obs])


def build_coal_graph(n):
    @with_ipv([n] + [0]*(n-1))
    def coal(state):
        T = []
        for i in range(state.size):
            for j in range(i, state.size):
                same = int(i == j)
                if same and state[i] < 2: continue
                if not same and (state[i]<1 or state[j]<1): continue
                new = state.copy(); new[i]-=1; new[j]-=1; new[i+j+1]+=1
                T.append((new, [state[i]*(state[j]-same)/(1+same)]))
        return T
    return Graph(coal)


def run_svgd_robust(graph, sparse_data, rewards, theta_start,
                    n_particles=60, n_iterations=300,
                    prior_override=None):
    for lr in [0.05, 0.1, 0.25]:
        graph.update_weights([theta_start])
        if prior_override is not None:
            prior = prior_override
        else:
            try:
                prior = GaussPrior(ci=[theta_start * 0.1,
                                       theta_start * 10])
            except Exception:
                prior = GaussPrior(ci=[theta_start * 0.1,
                                       theta_start * 10])

        sv = graph.svgd(
            sparse_data,
            rewards=rewards,
            prior=prior,
            optimizer=Adam(lr),
            n_particles=n_particles,
            n_iterations=n_iterations,
            return_history=True,
        )
        res = sv.get_results()
        std  = res['theta_std'].item()
        mean = res['theta_mean'].item()

        if std < 1e-8:
            print(f"  lr={lr}: KOLLAPS (std={std:.2e}) — prøver lavere lr")
            continue
        if mean < 0:
            print(f"  lr={lr}: NEGATIV mean prøver lavere lr")
            continue

        print(f"  lr={lr}: OK  theta={mean:.8f} ± {std:.8f}")
        return sv, res

    print("  ADVARSEL: alle lr gav problemer returnerer sidste kørsel")
    return sv, res

def build_two_island(n):
    idx = StateIndexer(descendants=[
        Property('pop1', min_value=0, max_value=n),
        Property('pop2', min_value=0, max_value=n),
        Property('in_pop', min_value=1, max_value=2),
    ])
    init = [0] * idx.state_length
    init[idx.descendants.props_to_index(pop1=1, pop2=0, in_pop=1)] = n

    @with_ipv(init)
    def two_isl(state):
        T = []
        if state[idx.descendants.indices()].sum() <= 1:
            return T
        for i in range(idx.descendants.state_length):
            if state[i] == 0: continue
            pi = idx.descendants.index_to_props(i)
            for j in range(i, idx.descendants.state_length):
                if state[j] == 0: continue
                pj = idx.descendants.index_to_props(j)
                if pj.in_pop != pi.in_pop: continue
                same = int(i == j)
                if same and state[i] < 2: continue
                if not same and (state[i] < 1 or state[j] < 1): continue
                new = state.copy(); new[i] -= 1; new[j] -= 1
                new[idx.descendants.props_to_index(
                    pop1=pi.pop1+pj.pop1, pop2=pi.pop2+pj.pop2,
                    in_pop=pi.in_pop)] += 1
                T.append((new, [state[i]*(state[j]-same)/(1+same), 0]))
            # Migration: fra in_pop=pi.in_pop til den anden population
            if state[i] > 0:
                new_pop = 3 - pi.in_pop  # 1→2 eller 2→1
                new = state.copy(); new[i] -= 1
                new[idx.descendants.props_to_index(
                    pop1=pi.pop1, pop2=pi.pop2, in_pop=new_pop)] += 1
                T.append((new, [0, state[i]]))
        return T
    return Graph(two_isl)

Asymmetrisk migration: P. cynocephalus + P. kindae

Fra NB03: Asymmetrisk migration (\(m_{12} \neq m_{21}\)) er identifiability ved \(n=4\) og N_OBS ≥ 3000. Med \(n=2\) er begge migrationsrater confunderede (NB03, fig-asym4-pairwise).

Jeg tester om genflow er retningsbestemt mellem de to populationer:

  • \(m_{12}\): migration fra P. cynocephalus til P. kindae
  • \(m_{21}\): migration fra P. kindae til P. cynocephalus

Fortolkning (NB03): Hvis CI’erne for \(m_{12}\) og \(m_{21}\) overlapper nul, kan jjeg ikke skelne fra nul-migration modellen er ikke identifiability for disse data. Hvis en retning er signifikant større, indikerer det asymmetrisk introgression.

def build_two_island_asym(n):
    idx = StateIndexer(descendants=[
        Property('pop1', min_value=0, max_value=n),
        Property('pop2', min_value=0, max_value=n),
        Property('in_pop', min_value=1, max_value=2),
    ])
    init = [0]*idx.state_length
    init[idx.descendants.props_to_index(pop1=1, pop2=0, in_pop=1)] = n

    @with_ipv(init)
    def two_asym(state):
        T = []
        if state[idx.descendants.indices()].sum() <= 1: return T
        for i in range(idx.descendants.state_length):
            if state[i] == 0: continue
            pi = idx.descendants.index_to_props(i)
            for j in range(i, idx.descendants.state_length):
                if state[j] == 0: continue
                pj = idx.descendants.index_to_props(j)
                if pj.in_pop != pi.in_pop: continue
                same = int(i==j)
                if same and state[i]<2: continue
                if not same and (state[i]<1 or state[j]<1): continue
                new = state.copy(); new[i]-=1; new[j]-=1
                new[idx.descendants.props_to_index(
                    pop1=pi.pop1+pj.pop1, pop2=pi.pop2+pj.pop2,
                    in_pop=pi.in_pop)] += 1
                T.append([new, [state[i]*(state[j]-same)/(1+same), 0, 0]])
            if pi.in_pop == 1:
                new = state.copy(); new[i]-=1
                new[idx.descendants.props_to_index(
                    pop1=pi.pop1, pop2=pi.pop2, in_pop=2)] += 1
                T.append([new, [0, state[i], 0]])
            else:
                new = state.copy(); new[i]-=1
                new[idx.descendants.props_to_index(
                    pop1=pi.pop1, pop2=pi.pop2, in_pop=1)] += 1
                T.append([new, [0, 0, state[i]]])
        return T
    return Graph(two_asym)

IM-modellen: P. anubis + P. cynocephalus

Fra NB05: IM-modellen er en tidsinhomogen coalescent med split-tidspunkt \(t_s\). Før \(t_s\) (i nutidig tid) eksisterer to populationer med migration; efter \(t_s\) (i fortiden) smelter de sammen til en ancestral population.

Kalibrering fra NB05:

Parameter Valg Begrundelse
Split-start ~1 Milioner år siden Rogers et al. (2019): nordlig/sydlig kladde divergens
\(t_s\) (skaleret) = split_yr / (2 * Ne * GEN) NB05: skalering fra år til coalescent-enheder
Prior (\(\theta_0\)) HalfCauchyPrior NB05: positiv-støttet, tunge haler
Prior (\(\theta_1\)) HalfCauchyPrior NB05: migrationsrater
epoch_switch_rate 15 / \(t_{\text{coal}}\) NB05: konservativ epoch-kalibrering
n_iterations 350 NB05: IM kræver flere iterationer end to-ø

Fortolkning (NB05): - \(E[\text{TMRCA}]\) er højere i IM end i to-ø fordi inter-population par altid skal vente til post-split ancestral coalescent (NB05, fig-im-tmrca) - Flaskehals ved splittet øger variances (NB05: stor \(f\) giver stor Var[TMRCA]) - Split-tid er fast (fra Rogers et al. 2019); vi estimerer \(\theta_0\) og \(\theta_1\)

Forventet resultat: P. anubis og P. cynocephalus har dokumenteret hybridzone (x2 i fig-Kort).Jeg forventer en ikke-nul migrationsrate og split for ca. 1 Mya.

    ])
    init = [0]*idx.state_length
    init[idx.descendants.props_to_index(pop1=1, pop2=0, in_pop=1)] = n

    @with_ipv(init)
    def im_mod(state):
        T = []
        if state[idx.descendants.indices()].sum() <= 1: return T
        for i in range(idx.descendants.state_length):
            if state[i] == 0: continue
            pi = idx.descendants.index_to_props(i)
            for j in range(i, idx.descendants.state_length):
                if state[j] == 0: continue
                pj = idx.descendants.index_to_props(j)
                if pj.in_pop != pi.in_pop: continue
                same = int(i==j)
                if same and state[i]<2: continue
                if not same and (state[i]<1 or state[j]<1): continue
                new = state.copy(); new[i]-=1; new[j]-=1
                new[idx.descendants.props_to_index(
                    pop1=pi.pop1+pj.pop1, pop2=pi.pop2+pj.pop2,
                    in_pop=pi.in_pop)] += 1
                T.append((new, [state[i]*(state[j]-same)/(1+same), 0, 0]))
            # Migration
            if state[i] > 0:
                new_pop = 3 - pi.in_pop
                new = state.copy(); new[i] -= 1
                new[idx.descendants.props_to_index(
                    pop1=pi.pop1, pop2=pi.pop2, in_pop=new_pop)] += 1
                T.append((new, [0, state[i], 0]))
        return T
    return Graph(im_mod)

def df_to_joint_obs(df, n_sub=4, n_obs=1000, seed=42):
    rng = np.random.default_rng(seed)
    der = df['derived_count'].values.astype(int)
    tot = df['nr_samples'].values.astype(int)
    obs = []
    for idx in rng.integers(0, len(df), n_obs*50):
        if len(obs) >= n_obs: break
        n, ac = tot[idx], der[idx]
        if n < n_sub: continue
        k = rng.hypergeometric(ac, n-ac, n_sub)
        if k == 0 or k == n_sub: continue
        vec = [0]*n_sub; vec[k-1] = 1
        obs.append(vec)
    return obs

Samlet msprime-validering af alle modeller

Fra NB01–NB06: msprime-validering er det kritiske trin der bekræfter, at hele pipeline model + skalering + SVGD-inferens fungerer korrekt. Jeg følger den samme validerings-protokol som i NB01 (Kingman), NB03 (to-ø) og NB05 (IM):

  1. Tag estimerede parametre fra phasic-SVGD
  2. Simuler data fra msprime med disse parametre (konverteret til msprime-enheder)
  3. Kør SVGD på msprime-data
  4. Bekræft at sande parametre ligger inden for 95%-CI

Tre scenarier testes:

  1. Simpel coalescent — estimer \(\theta\) for en bavian-population
  2. To-ø model — estimer \(\theta\) og migration
  3. Tidsinhomogen 2-epoke model — estimer nutidigt og historisk \(N_e\)

Skalering msprime ↔︎ phasic (lært fra NB03):

  • \(N_e^\text{msprime} = 1/(2\theta_0)\) (skalerede enheder, haploid)
  • \(m^\text{msprime} = \theta_1 \times \theta_0\) (backward migrationsrate)
  • Tidskonvertering: \(t_\text{år} = t_\text{scaled} \times 2 N_e^\text{gen} \times g\)

# Byg 2-epoke coalescent-model
n_ep_val = 2

    lineage=[Property('ton', min_value=1, max_value=VAL_N_SAMP)]
)
all_pairs_val = partial(combinations_with_replacement, r=2)

@with_ipv([VAL_N_SAMP] + [0] * (VAL_N_SAMP - 1))
def coal_val3(state, indexer=None):
    T = []
    for i, j in all_pairs_val(indexer_val.lineage):
        p1 = indexer_val.lineage.index_to_props(i)
        p2 = indexer_val.lineage.index_to_props(j)
        same = int(i == j)
        if same and state[i] < 2: continue
        if not same and (state[i] < 1 or state[j] < 1): continue
        new = state.copy(); new[i] -= 1; new[j] -= 1
        k = indexer_val.lineage.props_to_index(
            ton=p1.ton + p2.ton)
        if k < indexer_val.lineage.state_length:
            new[k] += 1
        T.append([new, [state[i] * (state[j] - same) / (1 + same)]])
    return T
